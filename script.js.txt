async function chargerViewers() {
  const { data } = await supabase
    .from("viewers")
    .select("*")
    .order("captures", { ascending: false });

  const container = document.getElementById("viewer-list");
  container.innerHTML = "";

  data.forEach((v, index) => {
    const div = document.createElement("div");
    div.className = "viewer";
    div.innerHTML = `
      <h2>#${index + 1} – ${v.name}</h2>
      <p>🐟 Captures : ${v.captures}</p>
      <p>🎒 Inventaire : ${v.inventory?.join(", ") || "vide"}</p>
      <p>🏅 Succès : ${v.success?.join(", ") || "aucun"}</p>
      <p>📜 Quête : ${v.quest_name || "aucune"} (${v.quest_progress}/${v.quest_goal})</p>
    `;
    container.appendChild(div);
  });
}

document.getElementById("refresh-btn").addEventListener("click", chargerViewers);

chargerViewers();
